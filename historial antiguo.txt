/** Muestra la variación porcentual con color y flecha. */
function mostrarVariacion($anterior, $nuevo): string
{
    $variacion = variacionPorcentual($anterior, $nuevo);
    if ($variacion === null) {
        return '<span class="var-flat">—</span>';
    }
    if (abs($variacion) < 0.005) {
        return '<span class="var-flat">0,00 %</span>';
    }
    $clase = $variacion > 0 ? 'var-up' : 'var-down';
    $signo = $variacion > 0 ? '▲ +' : '▼ ';
    return '<span class="' . $clase . '">' . $signo . esc(number_format($variacion, 2, ',', '.')) . ' %</span>';
}
