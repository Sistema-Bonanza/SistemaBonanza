-- MODULO DE VENTAS - Sistema Bonanza
-- Version corregida de las sentencias de tablas de ventas
--
-- COMO USARLO:
--  1. Ejecuta CADA sentencia por separado (cursor dentro + Ctrl+Enter).
--  2. Empieza por la SECCION 0 (solo lectura) para saber en que estado
--     quedo tu base antes de cambiar nada.
--  3. Si una sentencia da "Duplicate column", "Duplicate key name",
--     "Duplicate constraint" o "Table already exists", significa que ESA
--     parte ya estaba aplicada: saltala y sigue con la siguiente.
--  4. Antes de ejecutar los ALTER y CREATE: no dejes abierta ninguna
--     transaccion de pruebas (haz ROLLBACK). Estos comandos confirman
--     en silencio lo que haya pendiente.
--
-- ERRORES QUE TENIAN LAS SENTENCIAS ORIGINALES (ya corregidos aqui):
--  a) detalle_ventas apuntaba a productos(id), pero la PK de productos
--     es id_producto -> la tabla NO se podia crear.
--  b) El nombre fk_detalle_producto ya lo usa detalle_compras. En MySQL
--     el nombre de una FK es unico en TODA la base de datos.
--  c) fk_detalle_venta tenia ON DELETE CASCADE: borrar una venta
--     borraria sus lineas en silencio. Las ventas no se borran, se anulan.
--  d) Faltaban los CHECK (la red de seguridad que si tienen las compras),
--     la FK de id_usuario_anula y el UNIQUE (id_venta, id_producto).
--  e) estado y fecha_venta aceptaban NULL.
-- =====================================================================

USE sistema_bonanza;

-- =====================================================================
-- SECCION 0: DIAGNOSTICO (solo lectura)
-- =====================================================================

-- 0.1 Que tablas existen
SHOW TABLES;
-- ESPERADO: ventas aparece. detalle_ventas aparece SOLO si la creaste
-- corrigiendo el error (a); con el SQL original no se pudo crear.

-- 0.2 Restricciones actuales de las tablas involucradas
SELECT TABLE_NAME, CONSTRAINT_TYPE, CONSTRAINT_NAME
FROM information_schema.TABLE_CONSTRAINTS
WHERE TABLE_SCHEMA = 'sistema_bonanza'
  AND TABLE_NAME IN ('ventas','detalle_ventas','movimientos','productos')
ORDER BY TABLE_NAME, CONSTRAINT_TYPE, CONSTRAINT_NAME;

-- 0.3 Si movimientos ya tiene la columna id_venta
SHOW COLUMNS FROM movimientos LIKE 'id_venta';
-- ESPERADO: 1 fila si el paso 3 ya se aplico; vacio si no.


-- =====================================================================
-- SECCION 1: CORREGIR LA TABLA ventas (ya existe)
-- =====================================================================

-- 1.1 estado y fecha_venta no deben aceptar NULL
ALTER TABLE ventas
    MODIFY fecha_venta TIMESTAMP  NOT NULL DEFAULT CURRENT_TIMESTAMP,
    MODIFY estado      TINYINT(1) NOT NULL DEFAULT 1 COMMENT '1 registrada, 0 anulada';

-- 1.2 FK de quien anula + CHECK (igual que en compras)
ALTER TABLE ventas
    ADD CONSTRAINT fk_ventas_usuario_anula FOREIGN KEY (id_usuario_anula)
        REFERENCES usuarios (id) ON UPDATE CASCADE ON DELETE RESTRICT,
    ADD CONSTRAINT chk_ventas_estado CHECK (estado IN (0, 1)),
    ADD CONSTRAINT chk_ventas_montos CHECK (subtotal >= 0 AND monto_iva >= 0 AND total >= 0),
    ADD CONSTRAINT chk_ventas_tasa   CHECK (tasa_iva >= 0 AND tasa_iva <= 100),
    ADD CONSTRAINT chk_ventas_total  CHECK (total = subtotal + monto_iva);


-- =====================================================================
-- SECCION 2: CREAR detalle_ventas (version corregida)
-- Si "Table already exists": pegame el resultado de
--   SHOW CREATE TABLE detalle_ventas;
-- y la ajustamos con ALTER en lugar de recrearla.
-- =====================================================================
CREATE TABLE detalle_ventas (
    id_detalle       INT           NOT NULL AUTO_INCREMENT,
    id_venta         INT           NOT NULL,
    id_producto      INT           NOT NULL,
    cantidad         INT           NOT NULL,           -- siempre en unidades
    precio_unitario  DECIMAL(10,2) NOT NULL,           -- copia historica de productos.precio_venta
    aplica_iva       TINYINT(1)    NOT NULL,           -- copia historica
    subtotal_linea   DECIMAL(12,2) NOT NULL,           -- cantidad x precio_unitario
    costo_unitario   DECIMAL(12,4) NULL,               -- opcional: ultimo costo, para ganancias en Reportes

    PRIMARY KEY (id_detalle),

    -- No tiene sentido vender dos veces el mismo producto en la misma venta.
    CONSTRAINT uq_detventa_venta_producto UNIQUE (id_venta, id_producto),

    CONSTRAINT fk_detventa_venta FOREIGN KEY (id_venta)
        REFERENCES ventas (id_venta) ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT fk_detventa_producto FOREIGN KEY (id_producto)
        REFERENCES productos (id_producto) ON UPDATE CASCADE ON DELETE RESTRICT,

    CONSTRAINT chk_detv_cantidad  CHECK (cantidad > 0),
    CONSTRAINT chk_detv_precio    CHECK (precio_unitario > 0),
    CONSTRAINT chk_detv_aplica    CHECK (aplica_iva IN (0, 1)),
    CONSTRAINT chk_detv_subtotal  CHECK (subtotal_linea = cantidad * precio_unitario),
    CONSTRAINT chk_detv_costo     CHECK (costo_unitario IS NULL OR costo_unitario >= 0)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;


-- =====================================================================
-- SECCION 3: movimientos -> id_venta  (tu sentencia, sin cambios)
-- =====================================================================
ALTER TABLE movimientos
    ADD COLUMN id_venta INT NULL AFTER id_usuario,
    ADD CONSTRAINT fk_movimiento_venta FOREIGN KEY (id_venta) REFERENCES ventas (id_venta);


-- =====================================================================
-- SECCION 4: productos -> stock nunca negativo  (tu sentencia, sin cambios)
-- =====================================================================
ALTER TABLE productos
    ADD CONSTRAINT chk_productos_stock CHECK (stock_actual >= 0);


-- =====================================================================
-- SECCION 5: VERIFICACION ESTRUCTURAL (solo lectura)
-- =====================================================================
SELECT TABLE_NAME, CONSTRAINT_TYPE, COUNT(*) AS cantidad
FROM information_schema.TABLE_CONSTRAINTS
WHERE TABLE_SCHEMA = 'sistema_bonanza'
  AND TABLE_NAME IN ('ventas','detalle_ventas','movimientos','productos')
GROUP BY TABLE_NAME, CONSTRAINT_TYPE
ORDER BY TABLE_NAME, CONSTRAINT_TYPE;
-- ESPERADO:
--   detalle_ventas  CHECK 5 | FOREIGN KEY 2 | PRIMARY KEY 1 | UNIQUE 1
--   movimientos     CHECK 3 | FOREIGN KEY 4 | PRIMARY KEY 1
--   productos       CHECK 1 | FOREIGN KEY 1 | PRIMARY KEY 1 | UNIQUE 1
--   ventas          CHECK 4 | FOREIGN KEY 3 | PRIMARY KEY 1
-- (Si en productos CHECK sale 0, el paso 4 no se aplico.)
